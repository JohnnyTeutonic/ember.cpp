cmake_minimum_required(VERSION 3.18)
project(ember_integrated_pick_diagnostic LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
if(NOT DEFINED EMBER_SOURCE)
  message(FATAL_ERROR "EMBER_SOURCE must point to the frozen source")
endif()
add_subdirectory("${EMBER_SOURCE}" ember EXCLUDE_FROM_ALL)
add_executable(integrated_pick_probe integrated_probe.cpp)
target_compile_features(integrated_pick_probe PRIVATE cxx_std_17)
target_compile_definitions(integrated_pick_probe PRIVATE HAS_CUDA)
target_include_directories(integrated_pick_probe PRIVATE "${EMBER_SOURCE}" "${PREPARED_INPUTS}")
target_link_libraries(integrated_pick_probe PRIVATE tinyllama_core nlohmann_json::nlohmann_json CUDA::cudart CUDA::cublas)
