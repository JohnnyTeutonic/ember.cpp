cmake_minimum_required(VERSION 3.18)
project(ordered_reduction_probe LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
add_subdirectory("${EMBER_SOURCE}" ember EXCLUDE_FROM_ALL)
add_executable(ordered_probe kernel_probe.cu)
target_compile_features(ordered_probe PRIVATE cxx_std_17)
target_compile_definitions(ordered_probe PRIVATE HAS_CUDA NDEBUG)
if(ORDERED_CANDIDATE)
 target_compile_definitions(ordered_probe PRIVATE ORDERED_CANDIDATE)
endif()
target_include_directories(ordered_probe PRIVATE "${EMBER_SOURCE}")
target_link_libraries(ordered_probe PRIVATE tinyllama_core CUDA::cudart CUDA::cublas nlohmann_json::nlohmann_json)
